{{- /* The next first Sunday of a month from today (today included), skipping any an officer has cancelled. site.js works it out again in the browser, so it stays right between builds. */ -}} {{- $today := time.AsTime (now.Format "2006-01-02") -}} {{- $thisMonth := time.AsTime (printf "%s-01" ($today.Format "2006-01")) -}} {{- $cancelled := partial "meetings-cancelled.html" . -}} {{- $out := "" -}} {{- range $add := slice 0 1 2 3 4 5 -}} {{- if not $out -}} {{- $first := $thisMonth.AddDate 0 $add 0 -}} {{- $sunday := $first.AddDate 0 0 (mod (sub 7 (int $first.Weekday)) 7) -}} {{- $iso := $sunday.Format "2006-01-02" -}} {{- if and (ge $iso ($today.Format "2006-01-02")) (not (in $cancelled $iso)) -}}{{- $out = $sunday -}}{{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
